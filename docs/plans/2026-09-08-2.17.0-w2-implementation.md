```js
#!/usr/bin/env node
// scripts/measure-tool-descriptions.mjs
/**
 * Measure tool-description bytes as the MCP client sees them: spawn the
 * built server against a scratch project, complete the initialize
 * handshake, call tools/list, sum description lengths. Black-box on
 * purpose — it measures what reaches the field, not what the source says.
 *
 *   node scripts/measure-tool-descriptions.mjs            # print JSON for both surfaces
 *   node scripts/measure-tool-descriptions.mjs --write    # rewrite docs/plans/tool-description-bytes-baseline.json
 */
import { spawnSync } from "node:child_process";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { fileURLToPath } from "node:url";

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
const ENTRY = path.join(root, "dist", "index.js");
const BASELINE = path.join(root, "docs", "plans", "tool-description-bytes-baseline.json");

export async function measureToolDescriptions(projectRoot, { fullSurface }) {
  const twiningDir = path.join(projectRoot, ".twining");
  fs.mkdirSync(twiningDir, { recursive: true });
  fs.writeFileSync(
    path.join(twiningDir, "config.yml"),
    `version: 1\nproject_name: measure\ntools:\n  mode: full\n  full_surface: ${fullSurface}\nanalytics:\n  metrics:\n    enabled: false\n`,
  );
  const requests = [
    { jsonrpc: "2.0", id: 1, method: "initialize", params: { protocolVersion: "2024-11-05", capabilities: {}, clientInfo: { name: "measure", version: "0" } } },
    { jsonrpc: "2.0", method: "notifications/initialized" },
    { jsonrpc: "2.0", id: 2, method: "tools/list", params: {} },
  ];
  const r = spawnSync("node", [ENTRY, "--project", projectRoot], {
    input: requests.map((m) => JSON.stringify(m)).join("\n") + "\n",
    encoding: "utf8",
    timeout: 20000,
    env: { ...process.env, TWINING_DASHBOARD_PORT: "0", TWINING_DASHBOARD_NO_OPEN: "1", TWINING_DISABLED: "" },
  });
  const line = r.stdout.split("\n").find((l) => l.includes('"id":2'));
  if (!line) throw new Error(`tools/list produced no response (status ${r.status}); stderr:\n${r.stderr.slice(-800)}`);
  const tools = JSON.parse(line).result.tools;
  const perTool = {};
  let total = 0;
  for (const t of tools) {
    const bytes = Buffer.byteLength(t.description ?? "", "utf8");
    perTool[t.name] = bytes;
    total += bytes;
  }
  return { total_bytes: total, tools: perTool };
}

async function main() {
  const out = { measured_at: new Date().toISOString(), server_version: JSON.parse(fs.readFileSync(path.join(root, "package.json"), "utf8")).version, surfaces: {} };
  for (const surface of ["default", "full"]) {
    const scratch = fs.mkdtempSync(path.join(os.tmpdir(), `twining-descr-${surface}-`));
    try {
      out.surfaces[surface] = await measureToolDescriptions(scratch, { fullSurface: surface === "full" });
    } finally {
      fs.rmSync(scratch, { recursive: true, force: true });
    }
  }
  if (process.argv.includes("--write")) {
    fs.writeFileSync(BASELINE, JSON.stringify(out, null, 2) + "\n");
    console.log(`wrote ${BASELINE}`);
  }
  console.log(JSON.stringify(out, null, 2));
}

if (process.argv[1] && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  main().catch((err) => { console.error(err); process.exit(1); });
}
```

```ts
   * `supersedes`. Index order is chronological (ULID), so a later claimant
   * of the same target overwrites an earlier one — last-wins parity with
   * housekeeping's superseded_backfill. Built lazily, once per why() call,
```

```ts
 * Git sync-state probe (W2 J-3; cross-host plan §1.7 hazards, §4.2 shape).
 * Read-only by construction: every call is a query; the working-tree probe

const DEFAULT_TTL_MS = 5_000;
const cache = new Map<string, { at: number; state: SyncState }>();

export function resetSyncStateCache(): void {
  cache.clear();

function gitQuery(cwd: string, args: string[]): string | undefined {
  try {
    return execFileSync("git", args, {
      cwd,
      encoding: "utf-8",
      timeout: 1500,
      stdio: ["ignore", "pipe", "ignore"],
    }).trim();
  } catch {
    return undefined;
  }
}

/** Porcelain v1 XY parse over the records subtree: `??` untracked; anything else modified/staged/deleted. */
export function parsePorcelain(text: string): { uncommitted: number; untracked: number } {
  let uncommitted = 0;
  let untracked = 0;
  for (const line of text.split("\n")) {
    if (line.length < 3) continue;
    if (line.startsWith("??")) untracked++;
    else uncommitted++;
  }
  return { uncommitted, untracked };
}

export function probeSyncState(projectRoot: string, opts: SyncProbeOptions): SyncState {
  const now = opts.now ?? Date.now;
  const state: SyncState = {
    is_git: false, branch: null, detached: false, upstream: null, ahead: null, behind: null,
    uncommitted_records: null, untracked_records: null, records_tracked: null,
    export_records: opts.exportRecords, last_records_commit: null, last_fetch_at: null,
    last_fetch_age_hours: null, shallow: null, worktree_local: opts.worktreeLocal ?? false,
    probed_at: new Date(now()).toISOString(),
  };
  // No mirror (export_records:false): there is nothing to judge — emit the flag only.
  if (!opts.exportRecords) return state;

  // Argument ORDER is pinned by the unit test below — reversed prints the branch twice.
  const head = gitQuery(projectRoot, ["rev-parse", "HEAD", "--abbrev-ref", "HEAD", "--is-shallow-repository", "--git-common-dir"]);
  if (!head) return state; // not a repo, git absent, or an empty repo — all "unknown"
  const [, abbrev, shallow, commonDirRaw] = head.split("\n");
  state.is_git = true;
  state.detached = abbrev === "HEAD";
  state.branch = state.detached ? null : (abbrev ?? null);
  state.shallow = shallow === "true";

  const upstream = gitQuery(projectRoot, ["rev-parse", "--abbrev-ref", "--symbolic-full-name", "@{u}"]);
  if (upstream) {
    state.upstream = upstream;
    const counts = gitQuery(projectRoot, ["rev-list", "--left-right", "--count", "HEAD...@{u}"]);
    if (counts) {
      const [a, b] = counts.split(/\s+/).map(Number);
      state.ahead = Number.isFinite(a) ? a! : null;
      state.behind = Number.isFinite(b) ? b! : null;
    }
  }

  if (fs.existsSync(path.join(projectRoot, ".twining", "records"))) {
    // check-ignore exit codes: 0 ignored, 1 not ignored, 128 error.
    try {
      execFileSync("git", ["check-ignore", "-q", ".twining/records"], { cwd: projectRoot, stdio: "ignore", timeout: 1500 });
      state.records_tracked = false;
    } catch (err) {
      state.records_tracked = (err as { status?: number }).status === 1 ? true : null;
    }
    if (state.records_tracked) {
      const porcelain = gitQuery(projectRoot, [
        "--no-optional-locks", "status", "--porcelain", "--untracked-files=all", "--", ".twining/records",
      ]);
      if (porcelain !== undefined) {
        const p = parsePorcelain(porcelain);
        state.uncommitted_records = p.uncommitted;
        state.untracked_records = p.untracked;
      }
      state.last_records_commit = gitQuery(projectRoot, ["log", "-1", "--format=%cI", "--", ".twining/records"]) || null;
    }
  }

  // FETCH_HEAD is per-worktree and absent in a linked worktree: read the common dir's.
  const commonDir = path.resolve(projectRoot, commonDirRaw ?? ".git");
  try {
    const st = fs.statSync(path.join(commonDir, "FETCH_HEAD"));
    state.last_fetch_at = st.mtime.toISOString();
    state.last_fetch_age_hours = Math.round(((now() - st.mtimeMs) / 3_600_000) * 10) / 10;
  } catch {
    // absent on a fresh or push-only clone: unknown, never "never fetched"
  }
  return state;
}

/** TTL-cached per projectRoot. Recomputed after the TTL — a push changes ahead/behind with HEAD unchanged, so no sha-keyed shortcut. */
export function getSyncState(projectRoot: string, opts: SyncProbeOptions): SyncState {
  const now = opts.now ?? Date.now;
  const ttl = opts.ttlMs ?? DEFAULT_TTL_MS;
  const hit = cache.get(projectRoot);
  if (hit && now() - hit.at < ttl) return hit.state;
  const state = probeSyncState(projectRoot, opts);
  cache.set(projectRoot, { at: now(), state });
  return state;
}

/** Actionable states only. Uncommitted and unpushed are reported SEPARATELY — a commit-only host must never be nagged to push. */
export function syncWarnings(s: SyncState): string[] {
  const out: string[] = [];
  if (!s.is_git || !s.export_records) return out;
  if (s.records_tracked === false) {
    out.push("The .twining/records tree is gitignored — nothing recorded on this host replicates to any other host.");
    return out;
  }
  const local = (s.uncommitted_records ?? 0) + (s.untracked_records ?? 0);
  if (local > 0) {
    out.push(`${local} record file(s) under .twining/records are not committed — invisible to every other host until committed and pushed (git add .twining/records && git commit).`);
  }
  if (!s.worktree_local && s.upstream && (s.ahead ?? 0) > 0) {
    out.push(`${s.ahead} commit(s) not pushed to ${s.upstream} — other hosts cannot pull them.`);
  }
  return out;
}

/** The unconditional empty-result caveat (≈33 tokens; §6.3 decision point: ≤ 40). */
export function emptyResultCaveat(s: SyncState | null | undefined): string {

  return ` This host reads only what has been committed and pulled — records written elsewhere since your last pull${age} are invisible.`;
}
```
